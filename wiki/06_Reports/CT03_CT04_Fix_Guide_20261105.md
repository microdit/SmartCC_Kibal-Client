### **2.2 Step 2: sip_client.dart 완전 수정**

**원래 API (expected):**
```dart
SIPClient client = SIPClient();
await client.init(SIPSettings(...));
client.registrationState.listen(...)
client.callState.listen(...)
```

**실제 API (flutter_sip 0.0.8):**
```dart
FlutterSip flutterSip = FlutterSip();
FlutterSip.init('http://sip-server:port');
await flutterSip.loginSip(
  userName: 'username',
  password: 'password',
  displayName: 'Display Name',
  sipIp: 'sip-server',
  port: '5060',
);
flutterSip.onLoginStatusChange((status) { ... });
```

**Adapter Implementation:**

Create `flutter_sip_adapter.dart`:

```dart
import 'package:flutter_sip/flutter_sip.dart';
import 'sip_config.dart';
import 'sip_session.dart';

class SIPClientAdapter {
  final FlutterSip _flutterSip = FlutterSip();
  SipConfig? _config;
  SipClientState _state = SipClientState.idle;
  
  final StreamController<RegistrationState> _registrationController = 
      StreamController<RegistrationState>.broadcast();
  final StreamController<CallState> _callController = 
      StreamController<CallState>.broadcast();
  
  SIPClientAdapter() {
    _flutterSip.onLoginStatusChange(_handleStatusChange);
  }
  
  Future<bool> init(SipConfig config) async {
    _config = config;
    final sipUrl = 'http://${config.domain}:${config.port}';
    FlutterSip.init(sipUrl);
    _state = SipClientState.connected;
    return true;
  }
  
  Future<bool> register() async {
    if (_config == null) return false;
    
    await _flutterSip.loginSip(
      userName: _config!.username,
      password: _config!.password,
      displayName: _config!.displayName ?? _config!.username,
      sipIp: _config!.domain,
      port: _config!.port.toString(),
    );
    
    // Wait for status change
    await Future.delayed(Duration(seconds: 2));
    return _state == SipClientState.registered;
  }
  
  Future<bool> unregister() async {
    await _flutterSip.logoutSip();
    _state = SipClientState.unregistered;
    return true;
  }
  
  Stream<RegistrationState> get registrationState => _registrationController.stream;
  Stream<CallState> get callState => _callController.stream;
  
  void _handleStatusChange(LoginStatusEnum status) {
    switch (status) {
      case LoginStatusEnum.LOGIN_STATUS_REGISTRING:
        _state = SipClientState.connecting;
        break;
      case LoginStatusEnum.LOGIN_STATUS_REGISTERED:
        _state = SipClientState.registered;
        break;
      case LoginStatusEnum.LOGIN_STATUS_UNREGISTERED:
        _state = SipClientState.unregistered;
        break;
      case LoginStatusEnum.LOGIN_STATUS_REGISTRATION_FAILED:
      case LoginStatusEnum.LOGIN_STATUS_NONE:
        _state = SipClientState.error;
        break;
    }
    
    _registrationController.add(_mapToRegistrationState(status));
  }
  
  RegistrationState _mapToRegistrationState(LoginStatusEnum status) {
    switch (status) {
      case LoginStatusEnum.LOGIN_STATUS_REGISTRING:
        return RegistrationState.registering;
      case LoginStatusEnum.LOGIN_STATUS_REGISTERED:
        return RegistrationState.registered;
      case LoginStatusEnum.LOGIN_STATUS_UNREGISTERED:
        return RegistrationState.unregistered;
      default:
        return RegistrationState.failed;
    }
  }
  
  Future<Session> call(String target) async {
    await _flutterSip.makeCall(
      phoneNo: target,
      displayName: target,
    );
    return Session(null);
  }
}
```

Then update `sip_client.dart` to use the adapter.

  sipIp: 'sip-server-ip',

// Logout

// Call
await FlutterSip().makeCall(
  phoneNo: 'target',
